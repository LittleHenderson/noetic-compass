def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True, allow_nan=False)
def digest(value):
    return sha256(canonical(value).encode()).hexdigest()
def identity(path):
    data = path.read_bytes()
    return {"bytes": len(data), "sha256": sha256(data).hexdigest()}
def read(name):
    return json.loads((DIRECTORY / name).read_bytes())
def scoped(root, relative):
    p = PurePosixPath(relative)
    assert type(relative) is str and str(p) == relative and not p.is_absolute()
    assert "\\" not in relative and ":" not in relative and all(part not in (".", "..") for part in p.parts)
    path = (root / relative).resolve()
    assert path.is_relative_to(root.resolve()) and path.is_file()
    return path

    for row in rows:
        assert identity(scoped(root, row["path"])) == {key: row[key] for key in ("bytes", "sha256")}, row["path"]

def answer_from_graph(view, operation, question):
    labels = next(row["node_labels"] for row in bundle["views"] if row["view_id"] == view)
    outgoing = {value: set() for value in labels.values()}
    for branch in bundle["branches"]:
        assert branch["predicate"] == "contains"
        outgoing[labels[branch["source_node_id"]]].add(labels[branch["target_node_id"]])
    if operation == "LIST_PARTS":
        return sorted(outgoing[question["idea"]])
    if operation == "FIND_SHARED_PART":
        first, second = question["a"], question["b"]
        common = outgoing[first] & outgoing[second]
        if second in outgoing[first]: common.add(second)
        if first in outgoing[second]: common.add(first)
        assert len(common) <= 1
        return next(iter(common)) if common else "none"
    assert operation == "REACHED_VIA"
    root, target = question["idea"], question["part"]
    if target in outgoing[root]:
        return "direct"
    for first in sorted(outgoing[root]):
        queue, visited = deque(outgoing[first]), set()
        while queue:
            item = queue.popleft()
            if item == target:
                return "via:" + first
            if item not in visited:
                visited.add(item)
                queue.extend(outgoing[item] - visited)
    return "absent"
