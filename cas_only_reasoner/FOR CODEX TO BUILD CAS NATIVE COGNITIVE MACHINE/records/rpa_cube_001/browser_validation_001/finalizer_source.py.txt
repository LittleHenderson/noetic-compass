from hashlib import sha256
import json
from pathlib import Path
import shutil

RECORD=ROOT/'records/rpa_cube_001/browser_validation_001'
OUTER=Path('C:/Users/wakil/OneDrive/Desktop/tksCAS/cas_only_reasoner/FOR CODEX TO BUILD CAS NATIVE COGNITIVE MACHINE/output/rpa_cube_001')
INNER=ROOT/'records/rpa_cube_001/delivery_001'
checksum=lambda p:sha256(p.read_bytes()).hexdigest()
read=lambda p:json.loads(p.read_text(encoding='utf-8'))
canonical=lambda v:json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False)+'\n'

def write(path,value):
    with path.open('x',encoding='utf-8',newline='\n') as handle:handle.write(canonical(value))

def main():
    renderer=checksum(ROOT/'tools/render_cube_rpa_view.py');adapter=checksum(ROOT/'cncm/rpa_cube_adapter.py')
    assert renderer=='c34e478d587767629fc8256fce3792db1126789dc9f59b2d99d39fbc158993f8'
