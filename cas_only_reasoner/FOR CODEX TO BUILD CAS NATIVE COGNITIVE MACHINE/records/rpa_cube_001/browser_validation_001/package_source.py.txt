checksum = lambda p: sha256(p.read_bytes()).hexdigest()
canonical = lambda value: json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False)+'\n'

def write(path, value):
    with path.open('x',encoding='utf-8',newline='\n') as handle:
        handle.write(value)

def copy_exact(source, destination):
    assert source.is_file() and not source.is_symlink()
    destination.parent.mkdir(parents=True,exist_ok=True)
    assert not destination.exists()
    shutil.copyfile(source,destination)
    assert source.read_bytes()==destination.read_bytes()
    return {'source':str(source.relative_to(ROOT)),'destination':str(destination.relative_to(RECORD)),'sha256':checksum(source),'size':source.stat().st_size}
