def _list(value, label, maximum):
    if type(value) is not list or len(value) > maximum:
        raise ValueError(label + ': expected bounded list')


def _integer(value, label, maximum=2**63-1, minimum=0):
    if type(value) is not int or not minimum <= value <= maximum:
        raise ValueError(label + ': expected bounded raw integer')


def _tree(value):
    pending, count, text_size = [(value,0)], 0, 0
    while pending:
        item, depth = pending.pop(); count += 1
        if count > 1000000 or depth > 64:
            raise ValueError('view structure exceeds bound')
        if type(item) is dict:
            if len(item) > 4096 or any(type(key) is not str for key in item):
                raise ValueError('invalid view object')
            text_size += sum(len(key.encode('utf-8')) for key in item)
            pending.extend((child,depth+1) for child in item.values())
        elif type(item) is list:
            _list(item,'view array',4096)
            pending.extend((child,depth+1) for child in item)
        elif type(item) is str:
            text_size += len(item.encode('utf-8'))
        elif type(item) is int:
            if item.bit_length() > 512: raise ValueError('view integer exceeds512bits')
        elif item is not None and type(item) is not bool:
            raise ValueError('view values must be exact JSON primitives')
        if text_size > MAX_INPUT_BYTES: raise ValueError('view text exceeds byte bound')


def _budget(value):
    _exact(value,('instructions_remaining','checker_calls_remaining'),'policy budget')
    for key in value: _integer(value[key],key)
