```python
TRANSIENT_CODES = {"SlowDown", "InternalError", "ServiceUnavailable", "RequestTimeout",
                   "RequestLimitExceeded", "Throttling", "ThrottlingException",
                   "500", "502", "503", "504"}

def wait(self, job, timeout, *, poll_interval=5.0):
    from botocore.exceptions import BotoCoreError
    require(timeout == shared.RESOURCE_PLAN["instance_seconds_max"], "PRACTICE_WAIT_CALLER")
    deadline = time.monotonic() + INSTANCE_SECONDS_MAX
    while True:
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            return {"worker_status": "REMOTE_TIMED_OUT"}
        try:
            outcome = super().wait(job, min(remaining, shared.RESOURCE_PLAN["instance_seconds_max"]),
                                   poll_interval=poll_interval)
        except (BotoCoreError, self._client_error) as error:
            if isinstance(error, self._client_error) and self._error_code(error) not in TRANSIENT_CODES:
                raise
            self._event("wait_poll_transient_error", "RETRYING",
                        error=type(error).__name__ + ":" + str(error)[:300])
            time.sleep(max(poll_interval, 15.0))
            continue
        if outcome.get("worker_status") != "REMOTE_TIMED_OUT":
            return outcome
```
