One fixture rule, stated once, because it decides what each counterexample can prove. An
authored pair that names a role on each side (`source:direct` -> `target:direct`) is a

def test_cb03_a_complete_binding_separates_the_swapped_pairing():
    a = _frames(EDGE_SWAP["scene_A"], "contains", "A")
    b = _frames(EDGE_SWAP["scene_B"], "contains", "B")
    assert [frame.edge_pairing for frame in a] == ["COMPLETE"]
    assert len(a) == 1, "a role-labelled scene is one frame carrying both edges"
    assert endpoint_sets(a) == endpoint_sets(b), "fixture precondition: same endpoint sets"
    assert relation_frame_pairs(a) == [("A", "B"), ("C", "D")]
    assert relation_frame_pairs(b) == [("A", "D"), ("C", "B")]
    assert frame_pairing_signature(a[0]) != frame_pairing_signature(b[0]), (
        "the pairing signature separates the two arrangements")
    assert bag_payload(a) == bag_payload(b), (
        "the endpoint bag cannot separate them: both reduce to sources {A, C} against "
        "targets {B, D}. A representation shaped like the bag passes this scene while "
        "losing it, which is the whole point of the fixture")


def test_cb03_an_endpoint_role_bag_collapses_a_repeated_unpaired_edge():
    """v1.1 3.1: keep every filler and every role name, still lose how many pairs there are."""
    edge = EDGE_SWAP["unpaired_scene"]["edges"][0]
    one = _frames({"scene": [edge]}, "contains", "one")
    two = _frames({"scene": [(edge[0], edge[1]), (edge[0], edge[1])]}, "contains", "two")
    assert len(two) == 2, "the list body is one frame per authored edge"
    assert two[0].bindings == two[1].bindings, "the two copies carry the same binding"
    assert all(frame.edge_pairing == "UNPAIRED" for frame in one + two)
    assert endpoint_sets(one) == endpoint_sets(two), "both copies hold the same endpoints"
    from cb1_relational_fixtures import naive_endpoint_role_bag as _bag

    def bag_edges(corpus):
        """The bag's own view: one source list and one target list per frame."""
        entry = _bag(corpus)[0]
        return entry["sources"], entry["targets"]

    assert set(bag_edges(one)[0]) == set(bag_edges(two)[0]) == {"A"}
    assert set(bag_edges(one)[1]) == set(bag_edges(two)[1]) == {"B"}
    assert one[0].bindings.count(("target:direct", "B")) == 1
    assert two[0].bindings.count(("target:direct", "B")) == 1
    assert [filler for _, filler in two[0].bindings].count("B") == 1, (
        "each frame holds one copy, so a per-frame bag reports one copy either way; which "
        "is exactly why the per-frame bag cannot carry the count")
    assert naive_role_name_multiset(one) != naive_role_name_multiset(two), (
        "the raw name multiset is the only thing that grows, and it does not say which "
        "frame the extra copy came from")
    assert len(two) == 2, (
        "the frame-preserving record keeps both copies as separate frames; the frame count "
        "is the multiplicity, and it is exactly what a per-frame bag drops")
    assert frame_association_signature(two) != frame_association_signature(one), (
        "the association signature separates them while the bag does not")
    uniform = [RelationFrame(frame_ref="c", predicate="contains",
                             bindings=two[0].bindings, edge_pairing="UNPAIRED"),
               RelationFrame(frame_ref="c", predicate="contains",
                             bindings=two[0].bindings, edge_pairing="UNPAIRED")]
    assert bag_payload(uniform) == bag_payload(two), (
        "the bag also cannot tell which frame each copy came from")

def test_cb05_two_events_with_the_same_people_do_not_collapse():
    fixture = scene_two_events_same_people()
    frames = _frames({"scene": fixture["scene"]}, "HELP", "e")
    assert len(frames) == 2, "two events are two frames, not one"
    assert frames[0].frame_ref != frames[1].frame_ref
    assert frame_association_signature(frames) != frame_association_signature(frames[:1]), (
        "dropping the second event must change the association record")
    merged = RelationFrame(frame_ref="e:merged", predicate="HELP",
                           bindings=list(frames[0].bindings) + list(frames[1].bindings))
    assert sha(merged.canonical()) not in (sha(frames[0].canonical()), sha(frames[1].canonical()))
    assert sorted(sources for sources, _ in
                  [(entry["sources"], entry["targets"])
                   for entry in naive_endpoint_role_bag(frames)]) != \
        sorted(sources for sources, _ in
               [(entry["sources"], entry["targets"])
                for entry in naive_endpoint_role_bag(frames[:1])]), (
        "the bag does change here, because the two events exchange which filler is the "
        "helper; the collapse case is the repeated-occurrence fixture above, where the "
        "endpoint sets really are identical")
    vocabulary = lambda corpus: (sorted(set(naive_role_name_multiset(corpus)["roles"])),
                                 sorted(set(naive_role_name_multiset(corpus)["fillers"])))
    assert vocabulary(frames) == vocabulary(frames[:1]), (
        "one event and two use the same role and filler vocabulary; only the counts grow")
    assert len(naive_role_name_multiset(frames)["fillers"]) == 4, (
        "so the multiset sizes are the only trace, and they do not say which event a "
        "binding belongs to")


def test_cb05_repeated_occurrence_in_the_same_role_keeps_multiplicity():
    fixture = scene_repeated_occurrence_same_role()
    frames = _frames({"scene": fixture["scene"]}, "contains", "r")
    bindings = list(frames[0].bindings)
    repeats = bindings.count(("target:direct", "Wheel"))
    assert repeats == 2, "the repeated occurrence is two bindings: %r" % (bindings,)
    wheel_slots = [index for index, pair in enumerate(bindings)
                   if pair == ("target:direct", "Wheel")]
    assert wheel_slots == [1, 3], wheel_slots
    assert frames[0].occurrence_refs[1] != frames[0].occurrence_refs[3], (
        "the two occurrences of Wheel in the same role carry distinct references")
    assert len(set(frames[0].occurrence_refs)) == len(bindings), (
        "every binding has its own occurrence reference")
    once = RelationFrame(frame_ref="r:0", predicate="contains",
                         bindings=[("source:direct", "Bicycle"), ("target:direct", "Wheel")])
    assert sha(frames[0].canonical()) != sha(once.canonical())
    assert endpoint_sets(frames) == endpoint_sets([once]), (
        "one occurrence and two have the same endpoint sets")
    assert naive_endpoint_role_bag(frames) != naive_endpoint_role_bag([once]), (
        "here the bag does see the multiset, because each side is role-named; the "
        "single-frame unpaired form above is where the bag loses it")

def _families():
    return {
        "bicycle": [{"id": "b1", "family": "bicycle", "underlying_graph": "g1",
                     "linked_views": ["technical", "analogy", "story"]},
                    {"id": "b2", "family": "bicycle", "underlying_graph": "g1",

        "kit": [{"id": "k1", "family": "kit", "underlying_graph": "g2",
                 "linked_views": ["technical", "analogy"]}],
        "cart": [{"id": "c1", "family": "cart", "underlying_graph": "g3",
                  "linked_views": ["technical"]}],
    }


def test_cb24_family_grouped_splits_audit_clean():

    plan.assign("cart", "validation")

    assert audit["families"] == 3 and audit["assigned"] == 3

def _public_view(target):
    """A correct builder's public view: identical for every private target."""
    return {"task_text": "What are the direct parts of Bicycle?",
            "public_options": ["Wheel", "Rim", "Handlebar"],
            "node_order": ["Bicycle", "Wheel", "Frame"],
            "choice_order": [0, 1, 2],
            "public_ids": ["n1", "n2", "n3"]}


def _leaky_public_view(target):
    """A failing builder: the public option order depends on the private answer."""
    view = dict(_public_view(target))
    view["choice_order"] = [1, 0, 2] if target.get("answer") == 0 else [0, 1, 2]
    return view

    assert report["distinct_public_digests"] == 1

def test_cb16_frame_scoped_scheme_separates_the_swapped_pairing():
    a = _frames(EDGE_SWAP["scene_A"], "contains", "A")
    b = _frames(EDGE_SWAP["scene_B"], "contains", "B")
    scheme = pairing_preserving_scheme()
    encoded_a, encoded_b = scheme.encode(a), scheme.encode(b)
    assert encoded_a["blocks"] == encoded_b["blocks"], (
        "recorded limitation: a scheme whose role paths name only the endpoint side SORTS "
        "the fillers by side, so two edges inside one frame collapse. Frame scope alone does "
        "not save it. A builder using this scheme must keep one frame per edge, or add the "
        "edge identity to the role path")
    split_a = _frames([("A", "B"), ("C", "D")], "contains", "sa")
    split_b = _frames([("A", "D"), ("C", "B")], "contains", "sb")
    assert scheme.encode(split_a)["blocks"] != scheme.encode(split_b)["blocks"], (
        "with one frame per edge the same scheme separates the two arrangements")

def test_cb16_dimensions_are_checked_against_the_reference_formulation():
    result = tpr_reconstruct(bindings=[("A", "r0"), ("B", "r1")],
                             filler_embedding={"A": [1.0, 0.0], "B": [0.0, 1.0]},
                             role_embedding={"r0": [1.0, 0.0, 0.0], "r1": [0.0, 1.0, 0.0]},

def test_cb19_a_zero_weight_zero_bias_reconstruction_returns_exactly_zero():
    result = tpr_reconstruct(bindings=[("A", "r0"), ("B", "r0")],
                             filler_embedding={"A": [3.0], "B": [-2.0]},
                             role_embedding={"r0": [1.0]}, weight=[[0.0]], bias=[0.0])
    assert result["z_hat"] == [0.0], "no hidden constant may appear from nowhere"
    assert result["summed_occurrences"] == [1.0], "3.0 + (-2.0) sums to 1.0"


def test_cb19_single_bias_application_is_measurable():
    common = dict(bindings=[("A", "r0")], filler_embedding={"A": [1.0]},
                  role_embedding={"r0": [1.0]}, weight=[[1.0]])
    once = tpr_reconstruct(bias=[5.0], **common)["z_hat"][0]
    twice = tpr_reconstruct(bias=[10.0], **common)["z_hat"][0]
    assert once == 6.0
    assert twice == 11.0
    assert twice - once == 5.0, "one bias tensor, applied once"

def test_cb20_sum_of_occurrences_is_order_invariant_but_weight_is_not():
    filler = {"A": [1.0, 0.0], "B": [0.0, 1.0]}
    role = {"r0": [1.0, 0.0], "r1": [0.0, 1.0]}
    forward = tpr_reconstruct(bindings=[("A", "r0"), ("B", "r1")], filler_embedding=filler,
                              role_embedding=role, weight=[[1.0] * 4], bias=[0.0])
    backward = tpr_reconstruct(bindings=[("B", "r1"), ("A", "r0")], filler_embedding=filler,
                               role_embedding=role, weight=[[1.0] * 4], bias=[0.0])
    assert forward["summed_occurrences"] == backward["summed_occurrences"]
    differing = tpr_reconstruct(bindings=[("A", "r0"), ("B", "r1")], filler_embedding=filler,
                                role_embedding=role, weight=[[0.0, 1.0, 2.0, 3.0]], bias=[0.0])
    assert differing["summed_occurrences"] == forward["summed_occurrences"]
    assert differing["z_hat"] != forward["z_hat"], (
        "the sum is order-invariant; the weight is what makes the result order-sensitive")

def _audited_view():
    """A learner view with every closure surface in a declared state.

    Present: what the learner legitimately receives. Masked: content deliberately
    withheld. Declared absent: audited and genuinely not there. A surface in none of the
    three is `UNAUDITED` and fails the audit.
    """
    present = {"task_text", "choice_order", "choice_membership", "node_order",
               "public_ids", "public_options", "auxiliary_features"}
    masked = ("mask_shape", "answer_field")
    absent = tuple(sorted(set(CLOSURE_SURFACES) - present - set(masked)))
    view = {"task_text": "What are the direct parts of Bicycle?",
            "choice_order": [0, 1, 2],
            "choice_membership": ["Wheel", "Rim", "Handlebar"],
            "node_order": ["Bicycle", "Wheel", "Frame"],
            "public_ids": ["n1", "n2", "n3"],
            "public_options": ["Wheel", "Rim", "Handlebar"],
            "auxiliary_features": ["degree_in", "degree_out"]}
    return view, masked, absent


def _answer(operation, *arguments, inventory):
    edges = {tuple(pair) for pair in inventory["contains"]}
    complete = inventory["completeness"] == "complete"
    if operation == "LIST_PARTS":
        if [target for source, target in edges if source == arguments[0]]:
            return "DIRECT"
        return "ABSENT" if complete else "UNKNOWN"
    if operation == "REACHED_VIA":
        source, target = arguments
        if (source, target) in edges:
            return "DIRECT"
        frontier, seen, depth = {source}, {source}, 0
        while frontier and depth < 8:
            frontier = {b for a, b in edges if a in frontier and b not in seen}
            seen |= frontier
            depth += 1
            if target in seen:
                return "INDIRECT"
        return "ABSENT" if complete else "UNKNOWN"
    raise AssertionError("unknown operation %r" % operation)


def _mapping_ok(fixture):
    technical = {tuple(pair) for pair in fixture["technical"]["contains"]}
    analogical = {tuple(pair) for pair in fixture["analogical"]["contains"]}
    rows = fixture.get("relation_rows") or []
    if not rows:
        return "REJECTED"
    if {tuple(row["technical"]) for row in rows} != technical:
        return "REJECTED"
    if {tuple(row["analogical"]) for row in rows} != analogical:
        return "REJECTED"
    node_map = {row["technical"]: row["analogical"] for row in fixture["node_role_rows"]}
    for source, target in technical:
        if node_map.get(source) is None or node_map.get(target) is None:
            return "REJECTED"
        if (node_map[source], node_map[target]) not in analogical:
            return "REJECTED"
    return "ACCEPTED"
