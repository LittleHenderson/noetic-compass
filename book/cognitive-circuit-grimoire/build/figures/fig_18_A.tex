% Figure 18-A — CCB-ADV block diagram.
% Rhythm charges the capacitor; association installs and flips the cue; the
% path-select block carries the gap lane (detection-only) above the identity
% lane; both converge on conversion.  The A/B loop leaves the board outline
% because ^9^O x ^5^P is refused -- it closes on the advertiser's bench.
\providecommand{\casU}{\textasciicircum}
\providecommand{\casS}[1]{\_\{#1\}}
\begin{tikzpicture}[
  font=\sffamily\scriptsize,
  spine/.style={text width=45mm, minimum height=8mm, font=\sffamily\scriptsize}
]

  % ================= main spine =========================================
  \node[nblock, spine] (rhy) at (-0.9,0)
    {\textbf{RHYTHM} \textperiodcentered\ oscillator MEDIA\\
     f = 3.0/wk, sustain 12 wk};

  \node[pblock, spine] (cap) at (-0.9,-1.75)
    {\textbf{CHARGE} \textperiodcentered\ capacitor C\_SEG\\
     \casnote{\casU5\casU P\casS{F4}|f}\\
     C = 9.50, leak = 0.04/wk};

  \node[cblock, spine] (assoc) at (-0.9,-3.65)
    {\textbf{ASSOCIATION}\\
     ASSOC $\beta$=2.0 $\to$ cue \casnote{\casU8\casU C\casS{F4}|A}\\
     FLIP $\beta$=1.2 $\to$ \casnote{\casU9\casU C\casS{F4}|f}};

  % ---- path select: gap lane over identity lane ------------------------
  \node[eblock, spine] (gap) at (-0.9,-6.00)
    {\textbf{GAP lane} \textperiodcentered\ $\beta$ = 2.5\\
     \casnote{\casU3\casU E\casS{F3}|f}\\
     raises gain on an existing aversion};

  \node[eblock, spine] (idty) at (-0.9,-7.80)
    {\textbf{IDENTITY lane} \textperiodcentered\ $\beta$ = 2.0\\
     \casnote{\casU2\casU E\casS{F4}|f}\\
     conversion charges its own bias};

  \node[oblock, spine] (conv) at (-0.9,-9.80)
    {\textbf{CONVERSION}\\
     D3 diode \casnote{\casU E$\to$\casU O}, Vf = 2.0\\
     in series with R\_PRICE = 4.0};

  \node[casterm, text width=41mm] (tx) at (-0.9,-11.40)
    {TRANSACTION \ \casnote{\casU9\casU O\casS{F5}|E}};

  \draw[flow] (rhy)   -- (cap);
  \draw[flow] (cap)   -- (assoc);
  \draw[flow] (assoc) -- (gap);
  \draw[flow] (gap)   -- (idty);
  \draw[flow] (idty)  -- (conv);
  \draw[flow] (conv)  -- (tx);

  % PATH SELECT bracket, on the left of the two lanes
  \draw[line width=0.5pt, draw=casNedge]
    (-3.55,-4.80) -- (-3.88,-4.80) -- (-3.88,-8.55) -- (-3.55,-8.55);
  \node[font=\sffamily\scriptsize\bfseries, rotate=90, anchor=south,
        text=casNedge] at (-3.98,-6.68) {PATH SELECT};

  % ================= endorser side branch ===============================
  \node[pblock, text width=23mm, minimum height=7mm,
        font=\sffamily\scriptsize] (endo) at (3.45,-2.35)
    {\textbf{endorser}\\ \casnote{\casU5\casU P\casS{F7}|T}};

  \draw[flow, rounded corners=3pt]
    (endo.south) -- (3.45,-3.35) -- (1.48,-3.35);

  % ================= detection-only stamp on the gap lane ===============
  \node[font=\sffamily\tiny\bfseries, fill=white, draw=casEedge,

    {loop closes off-board:\\ \casnote{\casU9\casU O $\times$ \casU5\casU P} refused};

\end{tikzpicture}
