% Figure 23-C -- concurrency derating 1 - k(N-1) for the four coupling values.
% Headroom above 1.0 is deliberate: it carries the crossing table.
\begin{tikzpicture}
\begin{axis}[casaxis,
    width=10.2cm, height=7.4cm,
    grid=major, axis on top,
    xlabel={$N$ --- live concurrent operations},
    ylabel={derating factor $1 - k(N-1)$},
    xmin=1, xmax=8, ymin=0, ymax=1.5,
    xtick={1,2,3,4,5,6,7,8},
    ytick={0,0.2,0.4,0.5,0.6,0.7,0.8,1.0},
    yticklabels={0,0.20,0.40,0.50,0.60,0.70,0.80,1.00},
    clip mode=individual,
  ]

  % ---- the region where the linear model stops meaning anything ----------
  \fill[casEfill] (axis cs:1,0) rectangle (axis cs:8,0.5);
  \draw[casEedge, line width=0.6pt, densely dashed]
    (axis cs:1,0.5) -- (axis cs:8,0.5);
  \node[anchor=west, align=left, inner sep=1.5pt, fill=casEfill,
        font=\sffamily\scriptsize\itshape, text=casEedge]
    at (axis cs:1.06,0.16) {model not valid ---\\shed, do not compute};

  % ---- the four coupling lines ------------------------------------------
  \addplot[black, no marks, line width=1.0pt, domain=1:8, samples=2]
    {1-0.05*(x-1)};
  \addplot[casCedge, no marks, dashed, line width=1.0pt, domain=1:8, samples=2]
    {1-0.15*(x-1)};
  \addplot[casEedge, no marks, dash dot, line width=1.1pt, domain=1:8, samples=2]
    {1-0.35*(x-1)};
  \addplot[casNedge, no marks, densely dotted, line width=1.3pt, domain=1:8,
    samples=2] {1-0.60*(x-1)};

  % ---- design floor ------------------------------------------------------
  \draw[black, densely dashed, line width=0.7pt]
    (axis cs:1,0.7) -- (axis cs:8,0.7);
  \node[fill=white, inner sep=1.5pt, font=\sffamily\scriptsize\itshape]
    at (axis cs:4.6,0.70) {design floor $= 0.70$};

  % ---- where each line crosses the floor ---------------------------------
  \addplot[only marks, mark=*, mark size=2pt, black]
    coordinates {(7,0.7) (3,0.7) (1.857,0.7) (1.5,0.7)};

  % ---- inline line labels ------------------------------------------------
  \node[anchor=west, font=\sffamily\scriptsize, fill=white, inner sep=1pt]
    at (axis cs:3.30,0.92) {$k = 0.05$};
  \node[anchor=west, font=\sffamily\scriptsize, fill=white, inner sep=1pt]
    at (axis cs:4.80,0.55) {$k = 0.15$};
  \node[anchor=west, font=\sffamily\scriptsize, fill=white, inner sep=1pt]
    at (axis cs:2.45,0.56) {$k = 0.35$};
  \node[anchor=west, font=\sffamily\scriptsize, fill=white, inner sep=1pt]
    at (axis cs:1.05,0.34) {$k = 0.60$};
  \draw[casNedge, line width=0.4pt] (axis cs:1.92,0.34) -- (axis cs:2.10,0.34);

  % ---- crossing table ----------------------------------------------------
  \node[anchor=north east, draw=casNedge, fill=white, line width=0.4pt,
        inner sep=2.5pt, align=left, font=\sffamily\scriptsize]
    at (axis cs:7.92,1.47)
    {\textbf{crosses the 0.70 floor at}\\
     $k = 0.05$:\; $N = 7.0$\\
     $k = 0.15$:\; $N = 3.0$\\
     $k = 0.35$:\; $N = 1.9$\\
     $k = 0.60$:\; $N = 1.5$};
\end{axis}
\end{tikzpicture}
