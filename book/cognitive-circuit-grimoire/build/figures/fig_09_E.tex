               dash pattern=on 2.4pt off 1.8pt},

  ref/.style={font=\sffamily\footnotesize\bfseries}]

% ================= L1 : five-loop coil, bumps facing outward ==========
\draw[coilw] (1.60,0.50) -- (1.60,0.68);

% ================= L2 : five-loop coil ================================

\node[ref, anchor=south] at (6.40,2.90) {L$_2$ = 800 h$\cdot\Omega_c$};

% ================= coupling flux -- no wire between them =============
% dashed body -- an arrow tip on a dashed path renders hollow, so each tip is
% drawn as a short solid stub on the tangent instead.
\draw[flux] (2.05,2.30) to[out=18,in=162] (5.95,2.30);
\draw[flux] (2.05,1.63) to[out=0,in=180]  (5.95,1.63);

  (5.790,1.630) -- (5.95,1.63);

  (5.798,0.911) -- (5.95,0.96);

\node[tny, align=center, fill=white, inner sep=2pt, draw=casNedge,
      line width=0.4pt, rounded corners=2pt] at (4.00,1.63)
  {coupling flux\\$k = 0.50$\\$M = k\sqrt{L_1 L_2} = 800$};

% ================= the driving rate in L1 ============================
\draw[-{Latex[length=2.4mm,width=1.9mm]}, line width=1.6pt, draw=casCedge]

% direction contrast, stated in words for the grayscale reader
\node[tny, anchor=north east, align=right] at (1.46,0.42) {driven\\\emph{down}};

% ================= caption box =======================================
\node[nblock, text width=92mm, align=left, anchor=north,
      font=\sffamily\scriptsize] at (4.00,-0.42) {%

  $V_2 = M \times di_1/dt = 800 \times 0.25 = 200$\,V$_c$ --- 8.3$\times$ the

  Set $di_1/dt = 0$ and $V_2 = 0$: a steady conviction transmits nothing.};
