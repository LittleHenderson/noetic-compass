% Figure 9-B -- Characteristic curve: RL current rise on L = 2,000 h.Ohm_c.
% Three path constraints. tau = L/R and I_final = V/R share a denominator, so
% raising R pulls the transient IN and the ceiling DOWN at the same time.
% Axes are absolute (hours, A_c): in normalised coordinates all three curves
% coincide, which would hide the entire point of the plate.
\begin{tikzpicture}
\begin{axis}[casaxis,
  width=9.5cm, height=6.4cm,
  xlabel={hours on path\ \ (\emph{not} calendar hours)},
  ylabel={$i(t)$,\ A$_c$},
  xmin=0, xmax=300, ymin=0, ymax=1.45,
  xtick={0,50,100,150,200,250,300},
  xticklabels={0,{50\\$\tau$},{100\\$2\tau$},{150\\$3\tau$},200,250,300},
  xticklabel style={align=center},
  ytick={0,0.30,0.60,0.90,1.00,1.20},
  yticklabels={0,0.30,0.60,0.90,{$I_{\max}$},1.20},
  legend style={font=\sffamily\scriptsize, draw=casNedge, fill=white,
                inner sep=2.5pt, row sep=0.4pt,
                at={(0.025,0.975)}, anchor=north west},
]

% --- 63.2 / 86.5 / 95.0 per-cent construction on the design-point curve --
\foreach \tv/\iv in {50/0.3792, 100/0.5190, 150/0.5701} {
  \addplot[draw=casNedge!70, dash pattern=on 1.6pt off 1.4pt, line width=0.35pt,
           forget plot] coordinates {(0,\iv) (\tv,\iv) (\tv,0)};
}

% --- I_max ceiling: above this the path clips and spills ------------------
\addplot[draw=black, dash pattern=on 3pt off 2pt, line width=0.6pt, forget plot,
         domain=0:300, samples=2] {1.0};

% --- what R = 20 would reach if the path could carry it ------------------
\addplot[draw=casEedge!55, dotted, line width=0.8pt, forget plot,
         domain=0:300, samples=120] {1.2*(1-exp(-x/100))};

% --- the three constraint settings ---------------------------------------
\addplot[domain=0:300, samples=200, line width=1.3pt, draw=casEedge,
         dash pattern=on 4pt off 2pt] {min(1.0, 1.2*(1-exp(-x/100)))};
\addlegendentry{$R=20$: $\tau=100$\,h,\ $I_{\mathrm{final}}=1.20\rightarrow$ clips}

\addplot[domain=0:300, samples=200, line width=1.2pt, draw=casCedge]
  {0.6*(1-exp(-x/50))};
\addlegendentry{$R=40$: $\tau=50$\,h,\ $I_{\mathrm{final}}=0.60$ (design pt.)}

\addplot[domain=0:300, samples=200, line width=1.0pt, draw=casOedge,
         dash pattern=on 1.2pt off 1.5pt] {0.3*(1-exp(-x/25))};
\addlegendentry{$R=80$: $\tau=25$\,h,\ $I_{\mathrm{final}}=0.30$}

% --- annotations ----------------------------------------------------------
\node[font=\sffamily\scriptsize\itshape, anchor=north east, fill=white,
      inner sep=1.5pt] at (axis cs:296,0.985) {clipped -- flow spills};

\node[font=\sffamily\scriptsize, anchor=south east, align=right,
      draw=casNedge, fill=white, line width=0.4pt, rounded corners=2pt,
      inner sep=2.5pt] at (axis cs:296,0.035)
  {design point reaches 63.2\% of $I_{\mathrm{final}}$ at $\tau$,\\
   86.5\% at $2\tau$, 95.0\% at $3\tau$};

\end{axis}
\end{tikzpicture}
