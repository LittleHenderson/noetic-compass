% --- Unicode glyph fallback for verbatim/texttt contexts ---------------
% \unimath{name} produces \ensuremath{\name} — a single math glyph using the
% math font (Latin Modern Math), which has full coverage of math-symbol
% codepoints that the system mono fonts (Consolas) lack. Used by
% postprocess.py to substitute U+2208 (∈), U+2209 (∉), U+2282 (⊂),
% U+2286 (⊆), U+2283 (⊃), U+2287 (⊇), U+2205 (∅), U+27E8 (⟨), U+27E9 (⟩),
% U+2218 (∘) inside Verbatim/texttt blocks.
\newcommand{\unimath}[1]{\ensuremath{\csname #1\endcsname}}
